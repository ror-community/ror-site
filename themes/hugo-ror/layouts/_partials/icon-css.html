{{- /* CSS for the site's icons, generated from the SVGs in assets/icons/ (see the README there). Returns a CSS string that head.html adds to site.css. Each icon class (fa-, plus icon- for material-* files) sets --icon to its SVG as a data: URL and --icon-w to its width in em. The ::before rule below paints currentColor through that mask, so the markup stays as with the Font Awesome webfont, and icons size and color like text. Each SVG is in the stylesheet once rather than inlined in every page. An unknown icon name gets no --icon-w, so it takes no space rather than showing a filled box. This CSS is deliberately not in hugo-ror.css: ror-app also loads hugo-ror.css, together with its own Font Awesome, and these ::before rules would replace its icons. */ -}} {{- $css := slice `/* Icons: see themes/hugo-ror/layouts/_partials/icon-css.html */ .fa,.fa-solid,.fa-brands,.icon-material-search{display:inline-block;font-style:normal;line-height:1} .fa::before,.fa-solid::before,.fa-brands::before,.icon-material-search::before{content:"";display:inline-block;width:var(--icon-w,0);height:1em;vertical-align:-0.125em;background-color:currentColor;-webkit-mask-image:var(--icon);mask-image:var(--icon);-webkit-mask-size:contain;mask-size:contain;-webkit-mask-repeat:no-repeat;mask-repeat:no-repeat;-webkit-mask-position:center;mask-position:center}` -}} {{- /* Font Awesome 4 names still used in templates and content. */ -}} {{- $aliases := dict "magnifying-glass" (slice "search") "arrow-up-right-from-square" (slice "external-link") -}} {{- range resources.Match "icons/*.svg" -}} {{- $name := path.BaseName .Name -}} {{- $svg := .Content | strings.TrimSpace -}} {{- $vb := index (findRESubmatch `viewBox="0 0 ([\d.]+) ([\d.]+)"` $svg 1) 0 -}} {{- $w := div (float (index $vb 1)) (float (index $vb 2)) -}} {{- $data := $svg | replaceRE `"` "'" | replaceRE "#" "%23" | replaceRE "<" "%3C" | replaceRE ">" "%3E" -}} {{- $prefix := cond (strings.HasPrefix $name "material-") "icon-" "fa-" -}} {{- $selectors := slice (printf ".%s%s" $prefix $name) -}} {{- range index $aliases $name -}} {{- $selectors = $selectors | append (printf ".fa-%s" .) -}} {{- end -}} {{- $css = $css | append (printf `%s{--icon:url("data:image/svg+xml,%s");--icon-w:%.4gem}` (delimit $selectors ",") $data $w) -}} {{- end -}} {{- return delimit $css "\n" -}}
